Unikaj spadających klocków

Gotowe skrypty do Roblox Studio (kamera top-down). Arena musi nazywać się Arena i leżeć w Workspace.
KameraTopDown LocalScript
Miejsce: StarterPlayer → StarterPlayerScripts
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer
local camera = workspace.CurrentCamera
camera.CameraType = Enum.CameraType.Scriptable

local WYSOKOSC = 60

RunService.RenderStepped:Connect(function()
    local postac = player.Character
    local korzen = postac and postac:FindFirstChild("HumanoidRootPart")
    if korzen then
        local pozycja = korzen.Position + Vector3.new(0, WYSOKOSC, 0)
        camera.CFrame = CFrame.lookAt(pozycja, korzen.Position, Vector3.new(0, 0, -1))
    end
end)
SpawnerKlockow Script
Miejsce: ServerScriptService
local Debris = game:GetService("Debris")
local arena = workspace.Arena
local losuj = Random.new()

local odstep = 1.2

while true do
    local klocek = Instance.new("Part")
    klocek.Size = Vector3.new(
        losuj:NextInteger(2, 8),
        losuj:NextInteger(2, 8),
        losuj:NextInteger(2, 8)
    )
    klocek.Color = Color3.fromHSV(losuj:NextNumber(), 0.8, 1)

    local x = losuj:NextNumber(-arena.Size.X / 2, arena.Size.X / 2)
    local z = losuj:NextNumber(-arena.Size.Z / 2, arena.Size.Z / 2)
    klocek.Position = arena.Position + Vector3.new(x, 40, z)
    klocek.Parent = workspace

    local wyladowal = false

    klocek.Touched:Connect(function(dotkniety)
        local humanoid = dotkniety.Parent:FindFirstChild("Humanoid")
        if humanoid then
            humanoid.Health = 0
        elseif dotkniety == arena and not wyladowal then
            wyladowal = true
            Debris:AddItem(klocek, 0.3)
        end
    end)

    Debris:AddItem(klocek, 6)

    task.wait(odstep)
    odstep = math.max(0.5, odstep - 0.02)
end
Leaderstats Script
Miejsce: ServerScriptService (punkty i rekord)
local Players = game:GetService("Players")

Players.PlayerAdded:Connect(function(gracz)
    local stats = Instance.new("Folder")
    stats.Name = "leaderstats"
    stats.Parent = gracz

    local punkty = Instance.new("IntValue")
    punkty.Name = "Punkty"
    punkty.Value = 0
    punkty.Parent = stats

    local rekord = Instance.new("IntValue")
    rekord.Name = "Rekord"
    rekord.Value = 0
    rekord.Parent = stats

    gracz.CharacterAdded:Connect(function(postac)
        punkty.Value = 0
        local humanoid = postac:WaitForChild("Humanoid")

        humanoid.Died:Connect(function()
            if punkty.Value > rekord.Value then
                rekord.Value = punkty.Value
            end
        end)
    end)
end)
Monety Script
Miejsce: ServerScriptService (moneta co 3 sekundy)
local Players = game:GetService("Players")
local Debris = game:GetService("Debris")
local arena = workspace.Arena
local losuj = Random.new()

while true do
    task.wait(3)

    local moneta = Instance.new("Part")
    moneta.Shape = Enum.PartType.Cylinder
    moneta.Size = Vector3.new(0.5, 4, 4)
    moneta.Orientation = Vector3.new(0, 0, 90)
    moneta.Color = Color3.fromRGB(255, 200, 0)
    moneta.Material = Enum.Material.Neon
    moneta.Anchored = true
    moneta.CanCollide = false

    local margines = 5
    local x = losuj:NextNumber(-arena.Size.X / 2 + margines, arena.Size.X / 2 - margines)
    local z = losuj:NextNumber(-arena.Size.Z / 2 + margines, arena.Size.Z / 2 - margines)
    moneta.Position = arena.Position + Vector3.new(x, 3, z)
    moneta.Parent = workspace

    local zebrana = false

    moneta.Touched:Connect(function(dotkniety)
        if zebrana then return end
        local gracz = Players:GetPlayerFromCharacter(dotkniety.Parent)
        if gracz then
            zebrana = true
            gracz.leaderstats.Punkty.Value += 1
            moneta:Destroy()
        end
    end)

    Debris:AddItem(moneta, 10)
end
